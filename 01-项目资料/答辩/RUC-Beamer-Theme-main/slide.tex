% PDF P9
\begin{frame}{代表案例：“造舟于河”}
    \begin{block}{《经义述闻·左传下》}
        讨论“造舟于河”：这里的“造”应训为\key{比次}，还是训为\key{至}？
    \end{block}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{itemize}
            \item 《正义》承认“比舟为梁”，却转训“造”为“至”。
            \item 王念孙提出：“造之言曹也，相比次之名也。”
            \item 问题被收束为可证的训释命题。
        \end{itemize}
        \column{0.48\textwidth}
        \begin{itemize}
            \item “造、次一声之转”。
            \item “䔏”有副、次之义。
            \item 薛综注《东京赋》“以舟相比次为桥”可作旁证。
        \end{itemize}
    \end{columns}
    \vspace{0.2cm}
    \begin{center}
        \pill{发疑} \quad \pill{设问} \quad \pill{取证} \quad \pill{释理} \quad \pill{结论}
    \end{center}
\end{frame}

% PDF P19
\begin{frame}{AI 的位置：可核对的辅助}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{可以辅助}
            \begin{itemize}
                \item 相似案例召回
                \item 证据片段匹配
