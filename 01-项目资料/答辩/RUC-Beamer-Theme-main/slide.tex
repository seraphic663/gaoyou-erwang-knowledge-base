\begin{frame}{为什么是高邮二王}
    \begin{itemize}
        \item 高邮王念孙、王引之是乾嘉学派代表人物，“王氏四种”集中呈现清代考据学成熟方法。
        \item 材料边界清楚：《广雅疏证》《经义述闻》《经传释词》《读书杂志》适合一年周期集中推进。
        \item 方法类型集中：讹字、通假、因声求义、联绵词、虚词释读、句法辨析等反复出现。
        \item 适合跨学科表达：传统小学、文献学问题可以转化为结构化数据、检索界面和 AI 辅助流程。
    \end{itemize}
    \vspace{0.2cm}
    \begin{center}
        \pill{清代考据} \quad \pill{训诂学} \quad \pill{经学解释} \quad \pill{数字人文}
    \end{center}
\end{frame}

\begin{frame}{古典人文学术与现代数智技术的融合}
    \begin{columns}[T]
        \column{0.31\textwidth}

\begin{frame}{代表案例：“造舟于河”}
    \begin{block}{争点}
        《经义述闻·左传下》讨论“造舟于河”：这里的“造”应训为\key{比次}，还是训为\key{至}？
    \end{block}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{itemize}
            \item 《正义》承认“比舟为梁”，却转训“造”为“至”。
            \item 王念孙提出：“造之言曹也，相比次之名也。”
            \item 问题被收束为可证的训释命题。
        \end{itemize}
        \column{0.48\textwidth}
        \begin{itemize}
            \item “造、次一声之转”。
            \item “䔏”有副、次之义。
            \item 薛综注《东京赋》“以舟相比次为桥”可作旁证。
        \end{itemize}
    \end{columns}
    \vspace{0.2cm}
    \begin{center}
        \pill{发疑} \quad \pill{立说} \quad \pill{取证} \quad \pill{释理} \quad \pill{裁断旧说}
    \end{center}
\end{frame}

\section{多模态与泛化}

\begin{frame}{这里的“多模态”是什么意思}
    \begin{block}{不是只做文本检索}
        本项目同时处理多种材料形态，并让它们在同一条案例中互相指向。
    \end{block}
    \begin{columns}[T]
        \column{0.33\textwidth}
        \begin{exampleblock}{文本}
            原典、引文、二王论述、现代整理说明。
        \end{exampleblock}
        \column{0.33\textwidth}
        \begin{exampleblock}{结构}
            字段、表关系、证据类型、过程步骤。
        \end{exampleblock}
        \column{0.33\textwidth}
        \begin{exampleblock}{界面}
            流程图、数据库页、详情页、引用展开、AI 草案。
        \end{exampleblock}
    \end{columns}
    \vspace{0.25cm}
    \begin{center}
        \pill{原文} \quad \pill{批注} \quad \pill{JSON} \quad \pill{SQLite} \quad \pill{网页} \quad \pill{AI 输出}
    \end{center}
\end{frame}

\begin{frame}{AI 的位置：可核对的辅助}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{可以辅助}
            \begin{itemize}
                \item 相似案例召回
                \item 证据片段匹配
                \item “立论—取证—释理—结论”草稿
                \item 提醒人工复核点
            \end{itemize}
        \end{block}
        \column{0.48\textwidth}
        \begin{block}{不能替代}
            \begin{itemize}
                \item 原文与版本核对
                \item 通假方向判断
                \item 词义与文意区分
                \item 最终学术结论
            \end{itemize}
        \end{block}
    \end{columns}
    \vspace{0.25cm}
    \begin{center}
        \key{AI 不做裁判，只做可追溯材料上的助手。}
    \end{center}
\end{frame}

        \begin{itemize}

            \item 课堂阅读训练平台
